# preflight107.com — robots policy
#
# Search engines (Google, Bing, etc.): allowed — we want to be findable.
# AI training crawlers: disallowed below. These are HONOR-BASED — reputable
# AI companies respect robots.txt; non-reputable scrapers ignore it entirely.
# For a marketing site this is the right tradeoff: keep us indexable in
# Google + Bing while opting out of LLM training datasets where possible.

# ── AI training / inference bots (Disallow) ─────────────────────────────────

# OpenAI — used to gather training data for GPT
User-agent: GPTBot
Disallow: /

# OpenAI — fetches pages when ChatGPT browses / uses search

Disallow: /

# OpenAI — explicit ChatGPT-User crawler
User-agent: ChatGPT-User
Disallow: /

# Anthropic — Claude's web access (separate from training)

Disallow: /

# Anthropic — older training crawler identifier
User-agent: anthropic-ai
Disallow: /

# Anthropic — Claude when fetching on a user's behalf
User-agent: Claude-Web
Disallow: /

User-agent: CCBot
Disallow: /

Disallow: /
